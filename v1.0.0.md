actual class AuthManager {

    actual suspend fun signUp(email: String, password: String) {
        SignUp.create(
            SignUp.CreateParams.Standard(
                emailAddress = email,
                password = password
            )
        )
    }

    actual suspend fun signIn(email: String, password: String) {
        SignIn.create(
            SignIn.CreateParams.Strategy.Password(
                identifier = email,
                password = password
            )
        )
    }

    actual suspend fun signOut() {
        Clerk.signOut()
    }

    actual fun observeUser(): Flow<AuthUser?> =
        Clerk.userFlow.map {
            it?.let { u -> AuthUser(u.id, u.emailAddresses.first().emailAddress) }
        }

    actual suspend fun signInWithGoogle() {

    }
}
